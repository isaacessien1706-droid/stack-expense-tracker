Complete C++17 Implementation (Pure Stack Operations)

This C++ program implements a custom templated Stack<T> class from scratch and uses pure push(), pop(), peek(), isEmpty(), and isFull() operations. 

#include <iostream>
#include <vector>
#include <string>
#include <iomanip>

struct ApprovalStamp {
    int level;
    std::string roleTitle;
    std::string approverName;
    std::string comment;
};

template <typename T>
class Stack {
private:
    std::vector<T> items;
    int maxCapacity;
public:
    Stack(int capacity = 4) : maxCapacity(capacity) {}
    bool isEmpty() const { return items.empty(); }
    bool isFull() const { return static_cast<int>(items.size()) >= maxCapacity; }
    int size() const { return static_cast<int>(items.size()); }
    int getCapacity() const { return maxCapacity; }

    void push(const T& item) {
        if (!isFull()) items.push_back(item);
    }
    T pop() {
        T topItem = items.back();
        items.pop_back();
        return topItem;
    }
    T peek() const { return items.back(); }
};

class ExpenseInvoiceWorkflow {
private:
    std::string invoiceTitle;
    double amount;
    Stack<ApprovalStamp> approvalStack;
    Stack<ApprovalStamp> tempStack;

    const std::vector<std::string> roles = {
        "1. Employee Submission",
        "2. Team Lead Sign-Off",
        "3. Finance Manager Audit",
        "4. Executive Director Sign-Off"
    };

public:
    ExpenseInvoiceWorkflow(const std::string& title, double amt)
        : invoiceTitle(title), amount(amt), approvalStack(4), tempStack(4) {}

    void approveNextLevel(const std::string& name, const std::string& note) {
        if (approvalStack.isFull()) {
            std::cout << "[OVERFLOW] Invoice already has all 4 approvals!\n";
            return;
        }
        int idx = approvalStack.size();
        approvalStack.push({idx + 1, roles[idx], name, note});
        std::cout << "[PUSH] " << roles[idx] << " signed by " << name << "\n";
    }

    void rejectAndStepBack(const std::string& reason) {
        if (approvalStack.isEmpty()) {
            std::cout << "[UNDERFLOW] No signatures on stack to revoke!\n";
            return;
        }
        ApprovalStamp revoked = approvalStack.pop();
        std::cout << "[POP] Revoked " << revoked.roleTitle << " | Reason: " << reason << "\n";
    }

    void displayApprovalChain() {
        std::cout << "\n--- INVOICE APPROVAL STACK (" << approvalStack.size() << "/4) ---\n";
        bool isTop = true;
        while (!approvalStack.isEmpty()) {
            ApprovalStamp s = approvalStack.pop();
            tempStack.push(s);
            std::cout << (isTop ? "[TOP]    " : "[LOCKED] ") << s.roleTitle
                      << " | " << s.approverName << " (\"" << s.comment << "\")\n";
            isTop = false;
        }
        while (!tempStack.isEmpty()) {
            approvalStack.push(tempStack.pop());
        }
        std::cout << "------------------------------------------\n\n";
    }
};

int main() {
    ExpenseInvoiceWorkflow invoice("Server Rack Upgrade", 4500.00);
    invoice.approveNextLevel("Kwame Mensah", "Attached vendor quote");
    invoice.approveNextLevel("Ama Osei", "Verified department budget");
    invoice.approveNextLevel("Kofi Boateng", "VAT compliance checked");
    invoice.displayApprovalChain();

    invoice.rejectAndStepBack("Recalculate 2.5% levy before Director sign-off");
    invoice.displayApprovalChain();

    invoice.approveNextLevel("Kofi Boateng", "Levy updated and verified");
    invoice.approveNextLevel("Esi Danquah", "Executive payout authorized");
    invoice.displayApprovalChain();
    return 0;
}
